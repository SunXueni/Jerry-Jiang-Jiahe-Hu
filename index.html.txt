                    <ul class="feature-list">
                        <li>Concentrates solar energy with parabolic reflectors and vacuum solar collector tubes</li>

                        <li>Water is pumped to black metal heating tubes to generate steam, which is liquefied and collected in trapezoidal condensation chambers</li>

        function animateSun() {
            const centerX = window.innerWidth - 150;
            const centerY = 150;
            
            // Calculate sun position (follows mouse with limitations)

            sunY += (mouseY - centerY - sunY) * 0.05;
            
            // Limit movement range
            const maxMove = 80;
            sunX = Math.max(-maxMove, Math.min(maxMove, sunX));
            sunY = Math.max(-maxMove, Math.min(maxMove, sunY));
            
            sun.style.transform = `translate(${sunX}px, ${sunY}px)`;
            
            requestAnimationFrame(animateSun);
        }
